import { Card, CardGrid, LinkCard } from "@astrojs/starlight/components";

![The calendar editor, with one card per calendar and an overview of all directions](../../../../docs/images/editor.png)

<CardGrid>
  <Card title="Moved meetings move their block" icon="approve-check">
    Each occurrence of a recurring event is mirrored on its own. A moved or cancelled occurrence
    moves or removes its block.
  </Card>
  <Card title="Changes within seconds" icon="rocket">
    Microsoft and Google notify your deployment when a calendar changes. A periodic sync catches
    anything a notification missed.
  </Card>
  <Card title="Your tokens stay with you" icon="padlock">
    You run it in your own Cloudflare account. Tokens are encrypted at rest, and the EU data
    location is one setting.
  </Card>
  <Card title="Detail per direction" icon="setting">
    Per calendar and target, choose: not at all, as busy, with title, or everything. An example
    event shows the result before you save.
  </Card>
</CardGrid>
